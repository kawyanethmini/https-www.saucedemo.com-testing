package I_Assingment;

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;
import org.testng.Assert;
import org.testng.annotations.AfterClass;
import org.testng.annotations.BeforeClass;
import org.testng.annotations.Test;

import java.time.Duration;

public class Checkout {
    WebDriver driver;
    WebDriverWait wait;

    @BeforeClass
    public void setup() {
        driver = new ChromeDriver();
        driver.manage().window().maximize();
        driver.get("https://www.saucedemo.com/");

        wait = new WebDriverWait(driver, Duration.ofSeconds(10));
    }

    @Test(priority = 1)
    public void testLoginAndLandingPage() {
        driver.findElement(By.id("user-name")).sendKeys("standard_user");
        driver.findElement(By.id("password")).sendKeys("secret_sauce");
        driver.findElement(By.id("login-button")).click();

        WebElement header = driver.findElement(By.className("title"));
        String headerText = header.getText();

        Assert.assertEquals(headerText, "Products", "TC_01 Failed: Not on Products page!");

    }

    @Test(priority = 2)
    public void testHandleSecurityPopUp() {

        try {
            WebDriverWait alertWait = new WebDriverWait(driver, Duration.ofSeconds(5));
            alertWait.until(ExpectedConditions.alertIsPresent());

            String alertText = driver.switchTo().alert().getText();
            driver.switchTo().alert().accept();

        } catch (org.openqa.selenium.TimeoutException e) {
        }
        WebElement inventoryContainer = driver.findElement(By.id("inventory_container"));
        Assert.assertTrue(inventoryContainer.isDisplayed(), "TC_02 Failed: Inventory UI is blocked or not visible.");

    }
    @Test(priority = 3)
    public void testAddToCartAndBadge() {
        WebElement addBtn = wait.until(ExpectedConditions.elementToBeClickable(By.id("add-to-cart-sauce-labs-backpack")));
        addBtn.click();

        addBtn = wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("remove-sauce-labs-backpack")));

        String currentButtonText = addBtn.getText();

        Assert.assertEquals(currentButtonText, "Remove", "TC_06 Failed: Button caption did not update!");

        WebElement badge = driver.findElement(By.className("shopping_cart_badge"));
        String badgeCount = badge.getText();

        Assert.assertEquals(badgeCount, "1", "TC_06 Failed: Badge count is incorrect!");

        try { Thread.sleep(2000); } catch (InterruptedException e) { e.printStackTrace(); }
    }

    @Test(priority = 4)
    public void testCartConsistency() {
        WebElement cartLink = wait.until(ExpectedConditions.elementToBeClickable(By.className("shopping_cart_link")));
        cartLink.click();
        WebElement itemNameElement = wait.until(ExpectedConditions.visibilityOfElementLocated(By.className("inventory_item_name")));

        String itemQ = driver.findElement(By.className("cart_quantity")).getText();
        String itemName = itemNameElement.getText();
        String itemPrice = driver.findElement(By.className("inventory_item_price")).getText();

        Assert.assertEquals(itemQ, "1", "TC_07 Failed: Quantity mismatch!");
        Assert.assertEquals(itemName, "Sauce Labs Backpack", "TC_07 Failed: Item name mismatch!");
        Assert.assertEquals(itemPrice, "$29.99", "TC_07 Failed: Item price mismatch!");

        WebElement removeBtn = driver.findElement(By.id("remove-sauce-labs-backpack"));
        String buttonText = removeBtn.getText();
        Assert.assertEquals(buttonText, "Remove", "TC_07 Failed: Button caption is not 'Remove'!");


        try {
            Thread.sleep(3000);
        } catch (InterruptedException e) {
            throw new RuntimeException(e);
        }
    }
    @Test(priority = 5)
    public void testCheckoutInfo() {
        driver.findElement(By.id("checkout")).click();

        WebElement stepOneTitle = wait.until(ExpectedConditions.visibilityOfElementLocated(By.className("title")));
        System.out.println("Current Page Title: " + stepOneTitle.getText());

        driver.findElement(By.id("first-name")).sendKeys("Kawya");
        driver.findElement(By.id("last-name")).sendKeys("Nethmini");
        driver.findElement(By.id("postal-code")).sendKeys("12345");

        try { Thread.sleep(2000); } catch (InterruptedException e) { e.printStackTrace(); }

        driver.findElement(By.id("continue")).click();


        System.out.println("TC_09: Successfully moved from Information to Overview.");
        System.out.println("........................................................");
    }

    @Test(priority = 6)
    public void testPriceCalculation() {
        WebElement pageTitle = wait.until(ExpectedConditions.visibilityOfElementLocated(By.className("title")));
        System.out.println("Current Page: " + pageTitle.getText());

        String overviewName = driver.findElement(By.className("inventory_item_name")).getText();
        String overviewDesc = driver.findElement(By.className("inventory_item_desc")).getText();
        String overviewPrice = driver.findElement(By.className("inventory_item_price")).getText();

        Assert.assertEquals(overviewName, "Sauce Labs Backpack", "TC_10 Failed: Item Name mismatch on Overview!");
        Assert.assertFalse(overviewDesc.isEmpty(), "TC_10 Failed: Description is missing!");
        System.out.println("Overview Item Verified: " + overviewName + " | " + overviewPrice);

        String paymentInfo = driver.findElement(By.xpath("//div[@data-test='payment-info-value']")).getText();
        String shippingInfo = driver.findElement(By.xpath("//div[@data-test='shipping-info-value']")).getText();

        Assert.assertFalse(paymentInfo.isEmpty(), "TC_10 Failed: Payment Info missing!");
        Assert.assertFalse(shippingInfo.isEmpty(), "TC_10 Failed: Shipping Info missing!");

        System.out.println("Payment Info: " + paymentInfo);
        System.out.println("Shipping Info: " + shippingInfo);

        String subtotalTxt = driver.findElement(By.className("summary_subtotal_label")).getText();
        double itemTotal = Double.parseDouble(subtotalTxt.replaceAll("[^0-9.]", ""));

        String taxTxt = driver.findElement(By.className("summary_tax_label")).getText();
        double tax = Double.parseDouble(taxTxt.replaceAll("[^0-9.]", ""));

        String totalTxt = driver.findElement(By.className("summary_total_label")).getText();
        double actualTotal = Double.parseDouble(totalTxt.replaceAll("[^0-9.]", ""));

        double expectedTotal = itemTotal + tax;

        System.out.println("--- Calculation Breakdown ---");
        System.out.println("Subtotal: $" + itemTotal + " + Tax: $" + tax + " = $" + expectedTotal);

        Assert.assertEquals(actualTotal, expectedTotal, "TC_10 Failed: Total calculation mismatch!");

        System.out.println("TC_10 Passed: All Overview details and math are correct.");
        System.out.println("........................................................");
    }

    @Test(priority = 7)
    public void testOrderCompletion() {
        WebElement pageTitle = wait.until(ExpectedConditions.visibilityOfElementLocated(By.className("title")));
        WebElement finishBtn = wait.until(ExpectedConditions.elementToBeClickable(By.id("finish")));
        finishBtn.click();
        System.out.println("click finish button");

        WebElement finalTitle = wait.until(ExpectedConditions.visibilityOfElementLocated(By.className("title")));
        String pageTitleText = finalTitle.getText();
        System.out.println("Final Page Title: " + pageTitleText);

        Assert.assertEquals(pageTitleText, "Checkout: Complete!", "TC_11 Failed: Wrong page title after finishing!");

        WebElement successHeader = driver.findElement(By.className("complete-header"));
        String successMsg = successHeader.getText();
        System.out.println("Confirmation Message: " + successMsg);

        Assert.assertEquals(successMsg, "Thank you for your order!", "TC_11 Failed: Success message not found!");

        String dispatchTxt = driver.findElement(By.className("complete-text")).getText();
        System.out.println("Dispatch Note: " + dispatchTxt);

        System.out.println("TC_11 Passed: Order finalized and confirmation page verified.");
        System.out.println("........................................................");

        try { Thread.sleep(2000); } catch (InterruptedException e) { e.printStackTrace(); }
    }

    @Test(priority = 8)
    public void testReturnHome() {
        driver.findElement(By.id("back-to-products")).click();
        Assert.assertTrue(driver.getCurrentUrl().contains("inventory.html"), "TC_12 Failed: Return home failed!");
        System.out.println("TC_12 Passed: Successfully returned to the Home/Products page.");
        System.out.println("........................................................");
        try {
            Thread.sleep(3000);
        } catch (InterruptedException e) {
            throw new RuntimeException(e);
        }
    }

    @AfterClass
    public void tearDown() {
        if (driver != null) {
            driver.quit();
        }
    }
}
