package I_Assingment;

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.testng.Assert;
import org.testng.annotations.AfterClass;
import org.testng.annotations.BeforeClass;
import org.testng.annotations.Test;

import java.util.List;

public class CheckoutNegative {

    WebDriver driver;

    @BeforeClass
    public void setup() {
        driver = new ChromeDriver();
        driver.manage().window().maximize();
        driver.get("https://www.saucedemo.com/");
        
        driver.findElement(By.id("user-name")).sendKeys("standard_user");
        driver.findElement(By.id("password")).sendKeys("secret_sauce");
        driver.findElement(By.id("login-button")).click();
        try {
            Thread.sleep(5000);
        } catch (InterruptedException e) {
            throw new RuntimeException(e);
        }
    }


    @Test(priority = 1, description = "TC_13: Full Workflow for Empty Cart Checkout")
    public void testEmptyCartToFinish() {
        System.out.println("TC_13: Empty Cart Full Workflow ---");
        
        driver.findElement(By.className("shopping_cart_link")).click();
        System.out.println("Step 1: Entered Cart page. Cart is empty.");
        
        driver.findElement(By.id("checkout")).click();
        System.out.println("Step 2: Clicked Checkout button.");
        
        driver.findElement(By.id("first-name")).sendKeys("Test");
        driver.findElement(By.id("last-name")).sendKeys("User");
        driver.findElement(By.id("postal-code")).sendKeys("12345");
        driver.findElement(By.id("continue")).click();
        System.out.println("Step 3: Entered user info and clicked Continue.");
        
        driver.findElement(By.id("finish")).click();
        System.out.println("Step 4: Clicked Finish on the Overview page.");
        
        WebElement header = driver.findElement(By.className("complete-header"));
        String thanksMessage = header.getText();
        System.out.println("Step 5: Final Header Text: " + thanksMessage);
        boolean containsThankYou = thanksMessage.toUpperCase().contains("THANK YOU");
        
        Assert.assertTrue(containsThankYou, "Failed! Header did not contain 'THANK YOU'. Actual text: " + thanksMessage);
        Assert.assertTrue(driver.getCurrentUrl().contains("checkout-complete.html"), "URL did not match expected success page.");

        System.out.println("System allowed full checkout flow with an empty cart.");

    }

    @AfterClass
    public void tearDown() {
        if (driver != null) {
            driver.quit();

        }
    }
}
