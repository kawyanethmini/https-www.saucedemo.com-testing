package I_Assingment;

import org.openqa.selenium.By;
import org.openqa.selenium.Dimension;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;
import org.testng.Assert;
import org.testng.annotations.AfterClass;
import org.testng.annotations.BeforeClass;
import org.testng.annotations.Test;

import java.time.Duration;

public class NonFuntional {

    WebDriver driver;
    WebDriverWait wait;

    @BeforeClass
    public void setup() {
        driver = new ChromeDriver();
        driver.manage().window().maximize();
        wait = new WebDriverWait(driver, Duration.ofSeconds(10));
    }

    @Test(priority = 1, description = "NFC_02: Page Load Speed")
    public void testPageLoadSpeed() {
        long start = System.currentTimeMillis();

        driver.get("https://www.saucedemo.com/");
        wait.until(ExpectedConditions.elementToBeClickable(By.id("login-button")));
        long finish = System.currentTimeMillis();
        long totalTime = finish - start;
        System.out.println(" NFC_02: Page Load Time: " + totalTime + "ms");

        Assert.assertTrue(totalTime < 2000, "Performance Failure: Page took too long to load!");
    }

    @Test(priority = 2, description = "NF_01: Direct URL Access (Security)")
    public void testDirectUrlAccess() {
        driver.get("https://www.saucedemo.com/");
        driver.get("https://www.saucedemo.com/inventory.html");
        WebElement errorMsg = wait.until(ExpectedConditions.visibilityOfElementLocated(By.cssSelector("[data-test='error']")));
        String message = errorMsg.getText();

        Assert.assertTrue(message.contains("You can only access"), "Security Failure: Direct URL access was not blocked!");
        System.out.println("NFC_01: Security Check: Access Denied message displayed correctly.");
    }

    @Test(priority = 3, description = "NFC_03: Responsive Design (Usability)")
    public void testResponsiveDesign() {
        driver.manage().window().setSize(new Dimension(375, 812));
        driver.get("https://www.saucedemo.com/");
        WebElement loginBox = driver.findElement(By.className("login_wrapper"));
        Assert.assertTrue(loginBox.isDisplayed(), "Usability Failure: Login UI is broken on mobile width.");
        driver.manage().window().maximize();
        System.out.println("NFC_03: Usability Check: Mobile responsiveness validated.");
    }

    @AfterClass
    public void tearDown() {
        if (driver != null) {
            driver.quit();
        }
    }
}
