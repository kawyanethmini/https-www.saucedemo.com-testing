package I_Assingment;

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.support.ui.WebDriverWait;
import org.testng.Assert;
import org.testng.annotations.AfterClass;
import org.testng.annotations.BeforeClass;
import org.testng.annotations.Test;

import java.time.Duration;

public class CheckoutValidation {
    WebDriver driver;
    WebDriverWait wait;

    @BeforeClass
    public void setup() {
        driver = new ChromeDriver();
        driver.manage().window().maximize();
        wait = new WebDriverWait(driver, Duration.ofSeconds(10));
        driver.get("https://www.saucedemo.com/");

        driver.findElement(By.id("user-name")).sendKeys("standard_user");
        driver.findElement(By.id("password")).sendKeys("secret_sauce");
        driver.findElement(By.id("login-button")).click();
        try {
            Thread.sleep(5000);
        } catch (InterruptedException e) {
            throw new RuntimeException(e);
        }
    }
    
        @Test(priority = 1 , description = "TC_14: Missing Last Name Validation")
        public void testMissingLastName() {
            System.out.println("TC_14: Missing Last Name Validation");
            driver.findElement(By.id("add-to-cart-sauce-labs-backpack")).click();
            driver.findElement(By.className("shopping_cart_link")).click();
            driver.findElement(By.id("checkout")).click();

            driver.findElement(By.id("first-name")).sendKeys("Kawya");
            driver.findElement(By.id("postal-code")).sendKeys("10350");
            driver.findElement(By.id("continue")).click();
            System.out.println("Step: Submitted form with empty Last Name.");

            WebElement errorMsg = driver.findElement(By.cssSelector("[data-test='error']"));
            String errorText = errorMsg.getText();

            Assert.assertTrue(errorText.contains("Last Name is required"), "Error message mismatch!");
            System.out.println("System blocked checkout correctly.");
            System.out.println("................................");
        }

    @Test(priority = 2, description = "TC_15: Alphanumeric Zip Code Validation")
    public void testInvalidZipCodeFormat() {
        System.out.println("TC_15: Invalid Postal Code (ABCDE)");

        driver.get("https://www.saucedemo.com/inventory.html");
        System.out.println("Step: Navigated to Inventory page.");

        driver.findElement(By.id("add-to-cart-sauce-labs-bike-light")).click();
        System.out.println("Step: Item added to cart.");
        driver.findElement(By.className("shopping_cart_link")).click();
        driver.findElement(By.id("checkout")).click();
        System.out.println("Step: Click Checkout.");

        driver.findElement(By.id("first-name")).sendKeys("Kawya");
        driver.findElement(By.id("last-name")).sendKeys("Tester");
        driver.findElement(By.id("postal-code")).sendKeys("ABCDE");
        driver.findElement(By.id("continue")).click();
        System.out.println("Step: Submitted form with Postal Code as ABCDE.");

        String currentUrl = driver.getCurrentUrl();
        if (currentUrl.contains("checkout-step-two.html")) {
            System.out.println("[FAIL] Bug Detected! System accepted 'ABCDE'.");
            Assert.fail("Business Logic Bug: Zip Code field accepted non-numeric characters.");
        } else {
            System.out.println("RESULT: [PASS] System correctly blocked the invalid Zip Code.");
        }
    }

    @AfterClass
    public void tearDown() {
        if (driver != null) {
            driver.quit();
        }
    }


}
